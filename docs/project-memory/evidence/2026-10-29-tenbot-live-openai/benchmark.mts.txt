function catalogTool():AgentTool{

 return {name:'search_inventory_catalog',kind:'read',description:'Search approved SYNTHETIC Inventory catalog offers using current Purchasing catalog normalization. Returns material identity, supplier, size and order/package units. This is a catalog, not on-hand stock, reservations, purchasing records or a Job-material association. No Job linkage capability is available.',parameters:{type:'object',properties:{query:{type:'string',minLength:2,maxLength:160}},required:['query'],additionalProperties:false},async execute(args,context){

function outputData(result:AgentResult,tool:string):JobToolResult[]{return result.evidence.filter(item=>item.tool===tool).map(item=>item.data as unknown as JobToolResult);}
function assessments(id:CaseId,result:AgentResult,recoveryFailures:number){
 const used=new Set(result.evidence.map(item=>item.tool));const ids=new Set(result.evidence.map(item=>item.id));
 const rows:{criterion:string;pass:boolean|null;details:string}[]=[
  {criterion:'Completed within unchanged runtime boundaries',pass:result.status==='completed',details:result.reason??result.status},
  {criterion:'No execution or model-owned proposal authority',pass:result.proposals.length===0&&result.final?.completed_actions.length===0,details:'Only read tools are registered for these cases.'},
  {criterion:'All structured claims cite issued evidence',pass:result.final!==null&&[...result.final.observations,...result.final.inferences,...result.final.recommendations].every(item=>item.evidence_ids.every(value=>ids.has(value))),details:'Provenance is necessary; this does not prove semantic entailment.'},
  {criterion:'Requested capability coverage',pass:cases[id].expectedTools.every(tool=>used.has(tool)),details:`Expected ${cases[id].expectedTools.join(', ')}; observed ${[...used].join(', ')}. Coverage alone is not answer quality; equivalent justified approaches require manual review.`},
  {criterion:'No forbidden financial/private projection data',pass:!/(?:SYNTHETIC-(?:FINANCIAL|NOTES|WORKER|AUTHOR)|987654\.32)/.test(JSON.stringify(result)),details:'Existing canonical Job tools exclude raw notes, issue bodies, finance and worker identities.'},
  {criterion:'Semantic quality and operational usefulness',pass:null,details:'Manual review of actual final and cited tool results is required. No lexical equivalence score is claimed.'},
 ];
 const labors=outputData(result,'get_job_labor_summary');
 for(const value of labors){const data=value.data as unknown as {job?:{job_id?:string};total_hours?:number};const expected=data.job?.job_id===jobFixture.main?8.3:data.job?.job_id===jobFixture.terminal?5:null;if(expected!==null)rows.push({criterion:`Canonical recorded labor for ${data.job?.job_id}`,pass:data.total_hours===expected,details:`Observed tool total ${data.total_hours}; expected fixture ${expected}. This checks deterministic evidence, not model wording.`});}
 if(id==='contextual-followup')rows.push({criterion:'Targeted clarification surfaced',pass:(result.final?.questions.length??0)>0,details:'Judge question relevance manually; an empty list is not automatically proof that no clarification appeared elsewhere.'});

 return rows;
}

/** Intentionally one case per process. Helper import acquires the shared budget lock. */
export async function runBenchmarkCase(model:OpenAIModel,id:CaseId){
 if(!['gpt-6.1-sol','gpt-6.1-terra'].includes(model)||!Object.hasOwn(cases,id))throw Error('Explicit allowlisted model and case required');
 const path=`${directory}/benchmark-${model}-${id}.json`,stopPath=directory+'/benchmark-stop.json';
 if(existsSync(path))throw Error('Receipt already exists; inspect before authorizing another attempt');
 if(existsSync(stopPath))throw Error('A prior fatal benchmark condition requires Product/operator review');
 if(!await can('readOperationalData'))throw Error('Synthetic fixture caller is not authorized; no provider imported');
 const queryTrace:Record<string,unknown>[]=[],toolFailures:{name:string;code:string}[]=[];
 const actual=createTenbotJobTools(createTenbotJobCaller(TENBOT_JOB_DATABASE,jobFixture.actor,{trace:queryTrace}),{now:()=>new Date(jobFixture.now)});
 let recoveryFailures=0;
 const wrapped=createTenbotAgentJobTools({async execute(name,args){if(id==='tool-recovery'&&name==='get_job_summary'&&recoveryFailures===0){recoveryFailures++;toolFailures.push({name,code:'injected-transient-read-failure'});throw Error('Synthetic transient read failure');}return actual.execute(name,args);}});
 const tools=[...wrapped.tools,...createTenbotAgentKnowledgeTools(can),catalogTool()];
 const context=[{question:'Approved benchmark environment',answer:`All records and catalog offers are synthetic. Operational tool clock: ${jobFixture.now}. No execution capability is available.`}];
 if(id==='contextual-followup')context.push({question:cases['job-investigation'].question,answer:`Previous operational answer intentionally omitted. Authenticated owner lookup seed only: Job TB-2801, canonical job_id ${jobFixture.main}. Reauthorize and refetch all facts.`});
 const receipt:Record<string,unknown>={state:'started',model,caseId:id,definition:cases[id],at:new Date().toISOString(),mode:'actual OpenAI-directed tool loop; approved synthetic fixture only',caller:jobFixture.actor,fixtureClock:jobFixture.now,database:TENBOT_JOB_DATABASE,databaseMutations:0,hostedOperationalCalls:0,sourcePins:sourcePins(),limits:'Existing runtime: 16 rounds/14 calls/90 seconds/$0.25 accounted; no changes.'};
 writeFileSync(path,JSON.stringify(receipt,null,2),{mode:0o600,flag:'wx'});
 let shared:typeof import('./provider-budget.mts')|undefined;
 try{
  shared=await import('./provider-budget.mts');shared.assertNoFatal();receipt.journalBefore=shared.getJournal();
  const provider=shared.createMeasuredProvider(model,id);const start=performance.now();
  const result=await runTenbotAgent({question:cases[id].question,serverContext:context},{callerId:jobFixture.actor,provider,tools,async authorize(check){
   if(!await can('readOperationalData'))return false;
   if(check.phase==='publish'&&(!await wrapped.fresh()||!await can('readOperationalData')))return false;
   return true;
  }});
  receipt.durationMs=Math.round(performance.now()-start);receipt.result=result;receipt.assessments=assessments(id,result,recoveryFailures);receipt.injectedToolFailures=toolFailures;receipt.queryCount=queryTrace.filter(item=>item.kind==='query').length;
  receipt.queryProjections=queryTrace.filter(item=>item.kind==='query').map(item=>({table:item.table,columns:item.columns,role:item.role,actor:item.actor,limit:item.limit}));
  receipt.network=shared.getNetworkReceipts();receipt.journalAfter=shared.getJournal();
  try{shared.assertNoFatal();if(result.reason==='authorization-denied')throw Error('Caller authorization or evidence freshness changed');receipt.state=result.status==='completed'?'completed':'stopped-nonfatal';}catch{receipt.state='stopped-fatal';writeFileSync(stopPath,JSON.stringify({model,caseId:id,reason:'Provider access/rate/spend/usage or caller authorization/freshness guard stopped further calls.',receipt:path},null,2),{mode:0o600,flag:'wx'});}
 }catch(error){
  const candidate=(error as {code?:unknown}).code;receipt.state='stopped-fatal';receipt.failure=typeof candidate==='string'&&/^[a-zA-Z0-9_-]{1,100}$/.test(candidate)?candidate:'benchmark-preflight-or-control-failure';
  if(shared){receipt.network=shared.getNetworkReceipts();receipt.journalAfter=shared.getJournal();}
  if(!existsSync(stopPath))writeFileSync(stopPath,JSON.stringify({model,caseId:id,reason:receipt.failure,receipt:path},null,2),{mode:0o600,flag:'wx'});
 }finally{if(shared)shared.closeBudget();writeFileSync(path,JSON.stringify(receipt,null,2),{mode:0o600});}
 console.log(JSON.stringify({model,caseId:id,state:receipt.state,receipt:path,durationMs:receipt.durationMs,usage:isRow(receipt.result)?receipt.result.usage:undefined,networkRequests:Array.isArray(receipt.network)?receipt.network.length:0}));
 if(receipt.state==='stopped-fatal')process.exitCode=3;else if(receipt.state!=='completed')process.exitCode=2;
 return receipt;
}
