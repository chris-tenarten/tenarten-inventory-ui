begin read only;
set local statement_timeout='5s';
set local lock_timeout='1s';
select jsonb_build_object(
 'readOnly',current_setting('transaction_read_only'),
 'serverNow',clock_timestamp(),
 'databaseStartedAt',pg_postmaster_start_time(),
 'maxConnections',current_setting('max_connections')::int,
 'backendStates',(select jsonb_agg(s) from(select backend_type,coalesce(state,'unknown')state,count(*)n from pg_stat_activity where pid<>pg_backend_pid() group by backend_type,state order by backend_type,state)s),
 'clientBackends',(select count(*) from pg_stat_activity where backend_type='client backend' and pid<>pg_backend_pid()),
 'blockedBackends',(select count(*) from pg_stat_activity where pid<>pg_backend_pid() and cardinality(pg_blocking_pids(pid))>0),
 'longestClientQuerySeconds',(select coalesce(max(extract(epoch from clock_timestamp()-query_start)),0) from pg_stat_activity where pid<>pg_backend_pid() and backend_type='client backend' and state='active'),
 'longestClientTransactionSeconds',(select coalesce(max(extract(epoch from clock_timestamp()-xact_start)),0) from pg_stat_activity where pid<>pg_backend_pid() and backend_type='client backend' and xact_start is not null),
 'idleInTransactionClients',(select count(*) from pg_stat_activity where backend_type='client backend' and state like 'idle in transaction%'),
 'statementStatsView',(select n.nspname||'.'||c.relname from pg_class c join pg_namespace n on n.oid=c.relnamespace where c.relname='pg_stat_statements' and c.relkind='v' order by n.nspname limit 1),
 'scheduledJobs',(with k(jobid,name)as(values(1::bigint,'tenops-messaging-abandoned-drafts'),(2::bigint,'tenops-web-push'),(3::bigint,'tenbot-invoice-original-expiry')) select jsonb_agg(s)from(select k.jobid,k.name,j.jobid is not null as job_exists,j.active,j.schedule from k left join cron.job j on j.jobid=k.jobid and j.jobname=k.name order by k.jobid)s),
 'cronActivity',(with k(jobid,name)as(values(1::bigint,'tenops-messaging-abandoned-drafts'),(2::bigint,'tenops-web-push'),(3::bigint,'tenbot-invoice-original-expiry')),
 recent as materialized(select jobid,status,start_time,end_time,return_message from cron.job_run_details where jobid in(1,2,3) order by runid desc limit 200),
 w(label,since)as(values('since_verified_recovery','2026-10-30T23:55:40Z'::timestamptz),('last_30m',now()-interval'30 minutes'))
 select jsonb_agg(s)from(select w.label,k.jobid,k.name,count(r.jobid)started,count(r.jobid)filter(where r.status='failed')failed,count(r.jobid)filter(where r.status='succeeded')succeeded,count(r.jobid)filter(where r.end_time is null)unfinished,max(r.start_time)last_started,max(r.end_time)last_ended,max(extract(epoch from r.end_time-r.start_time))filter(where r.end_time is not null)max_finished_seconds,max(extract(epoch from now()-r.start_time))filter(where r.end_time is null)max_unfinished_record_age_seconds,count(r.jobid)filter(where r.status='failed' and r.return_message~*'connection|timeout|could not connect|failed to connect')connection_or_timeout_failures from w cross join k left join recent r on r.jobid=k.jobid and r.start_time>=w.since group by w.label,k.jobid,k.name order by w.label,k.jobid)s),
 'cronSampleCoverage',(select jsonb_build_object('limit',200,'sampledRows',count(*),'oldestStartedAt',min(start_time)) from(select start_time from cron.job_run_details where jobid in(1,2,3)order by runid desc limit 200)s),
 'invoiceCleanup',(with w(label,since)as(values('since_verified_recovery','2026-10-30T23:55:40Z'::timestamptz),('last_30m',now()-interval'30 minutes'))select jsonb_agg(s)from(select w.label,count(r.id)wakes,count(r.id)filter(where r.status='succeeded')succeeded,count(r.id)filter(where r.status='failed')failed,count(r.id)filter(where r.status='cancelled')cancelled,count(r.id)filter(where r.status='queued' and r.expires_at<extract(epoch from now()))expired_queued,count(r.id)filter(where r.status='running' and r.consumed_at<now()-interval'100 seconds')stale_running_records,coalesce(sum(r.removed_count),0)removed,max(r.created_at)last_wake,max(r.finished_at)last_finished,max(extract(epoch from r.finished_at-r.consumed_at))filter(where r.finished_at is not null and r.consumed_at is not null)max_finished_worker_seconds from w left join public.invoice_receival_cleanup_runs r on r.created_at>=w.since group by w.label order by w.label)s),
 'webPushWakes',(with w(label,since)as(values('since_verified_recovery','2026-10-30T23:55:40Z'::timestamptz),('last_30m',now()-interval'30 minutes')),e(environment)as(values('tenops'),('tendev'))select jsonb_agg(s)from(select w.label,e.environment,count(r.id)wakes,count(r.id)filter(where r.consumed_at is not null)consumed,count(r.id)filter(where r.consumed_at is null and r.expires_at<extract(epoch from now()))expired_unconsumed,max(r.created_at)last_wake,max(r.consumed_at)last_consumed,max(extract(epoch from r.consumed_at-r.created_at))filter(where r.consumed_at is not null)max_consumption_lag_seconds from w cross join e left join public.web_push_runs r on r.environment=e.environment and r.created_at>=w.since group by w.label,e.environment order by w.label,e.environment)s),
 'writeControls',jsonb_build_object('pilot',(select enabled from public.tenbot_receival_pilot_policy where singleton),'execution',(select execution_enabled from public.tenbot_receival_execution_control where singleton),'enrollment',(select count(*)from public.tenbot_receival_pilot_employees),'activeGrants',(select count(*)from public.tenbot_chris_dev_canary_grants where revoked_at is null and expires_at>clock_timestamp()),'permits',(select count(*)from public.invoice_receival_execution_permits))
)as state;
commit;
